# Meson would otherwise append -sPTHREAD_POOL_SIZE=4 from c_thread_count.
# That numeric pool is last on the link line and replaces the JavaScript
# expression set on the emulator. 0 leaves -pthread and no pool size.
c_thread_count = 0
cpp_thread_count = 0
c_args = ['-pthread']
cpp_args = ['-pthread']
objc_args = ['-pthread']
# -sPROXY_TO_PTHREAD link time flag always requires -pthread even during
# configuration so explicitly add the flag here.
# PTHREAD_POOL_SIZE is also set on the emulator link in meson.build, after
# dependency libraries. A .pc file that adds -sPTHREAD_POOL_SIZE would
# otherwise replace this one, because Emscripten keeps the last value.
c_link_args = ['-pthread','-sASYNCIFY=1','-sASYNCIFY_STACK_SIZE=65536','-sPROXY_TO_PTHREAD=1','-sOFFSCREENCANVAS_SUPPORT=1','-sDEFAULT_LIBRARY_FUNCS_TO_INCLUDE=$GL','-sMAX_WEBGL_VERSION=2','-sFORCE_FILESYSTEM','-sALLOW_TABLE_GROWTH','-sTOTAL_MEMORY=2GB','-sWASM_BIGINT','-sEXPORT_ES6=1','-sASYNCIFY_IMPORTS=ffi_call_js,emwgpuWaitAny','-sEXPORTED_RUNTIME_METHODS=addFunction,removeFunction,TTY,FS','-sPTHREAD_POOL_SIZE=(navigator.hardwareConcurrency||4)+8']
cpp_link_args = ['-pthread','-sASYNCIFY=1','-sASYNCIFY_STACK_SIZE=65536','-sPROXY_TO_PTHREAD=1','-sOFFSCREENCANVAS_SUPPORT=1','-sDEFAULT_LIBRARY_FUNCS_TO_INCLUDE=$GL','-sMAX_WEBGL_VERSION=2','-sFORCE_FILESYSTEM','-sALLOW_TABLE_GROWTH','-sTOTAL_MEMORY=2GB','-sWASM_BIGINT','-sEXPORT_ES6=1','-sASYNCIFY_IMPORTS=ffi_call_js,emwgpuWaitAny','-sEXPORTED_RUNTIME_METHODS=addFunction,removeFunction,TTY,FS','-sPTHREAD_POOL_SIZE=(navigator.hardwareConcurrency||4)+8']
